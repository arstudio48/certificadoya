MODIFICACIÓN DE js/main.js (lógica de cálculo)
═══════════════════════════════════════════════════════════════════════════════

Reemplazar función calcularPrecio():

ANTES:
  function calcularPrecio(provincia) {
    const basePrice = 57.00;
    const tasa = PROVINCIA_TASAS[provincia]?.tasa || 0.00;
    return basePrice + tasa;
  }

DESPUÉS:
  function calcularPrecio(provincia) {
    // PRECIOS ACTUALIZADOS +15% (28 julio 2026)
    const basePrice = 65.55;  // antes 57.00€
    const tasa = PROVINCIA_TASAS_2026[provincia]?.tasa || 0.00;
    const total = basePrice + tasa;
    
    // Actualizar UI
    document.getElementById('precio-base').textContent = basePrice.toFixed(2) + '€';
    document.getElementById('precio-tasa').textContent = tasa.toFixed(2) + '€';
    document.getElementById('precio-total').textContent = total.toFixed(2) + '€';
    document.getElementById('ccaa-nombre').textContent = provincia;
    
    // Mostrar/ocultar fila de tasa
    if (tasa > 0) {
      document.getElementById('tasa-row').style.display = 'block';
      const decreto = PROVINCIA_TASAS_2026[provincia].decreto;
      document.getElementById('tasa-tooltip').textContent = `(${decreto})`;
    } else {
      document.getElementById('tasa-row').style.display = 'none';
    }
    
    // Guardar para Stripe
    document.getElementById('hidden-precio-base').value = basePrice;
    document.getElementById('hidden-tasa-ccaa').value = tasa;
    document.getElementById('hidden-total').value = total;
    
    return total;
  }
